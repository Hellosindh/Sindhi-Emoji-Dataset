سنڌي ايموجي ڊيٽا (Sindhi Emoji Data) 🌟هي هيلوسنڌ (HelloSindh.com) پاران هڪ اوپن سورس پروجيڪٽ آهي، جنهن ۾ معياري يونيڪوڊ ايموجيز جو سنڌي ترجمو ۽ لاڳاپيل لفظ (Keywords) ڏنا ويا آهن.هن ڊيٽابيس جو مقصد سنڌي ٻوليءَ کي ڊجيٽل رابطن ۾ هٿي ڏيڻ آهي، ته جيئن ڊيولپرز آسانيءَ سان پنهنجي ويب يا موبائيل ايپليڪيشنز ۾ سنڌي ايموجي ڪيبورڊ، سرچ سسٽم ۽ ريئيڪشن فيچرز شامل ڪري سگهن.📦 پروجيڪٽ جون خاصيتونهن پروجيڪٽ جي سموري ڊيٽا emojis-data.js فائيل ۾ موجود آهي.  ٻه لساني سپورٽ (Bilingual Support): انگريزي (name_en, keywords_eng) ۽ سنڌي (name_sd, keywords_sd) ٻنهي ٻولين ۾ نالا ۽ لفظ موجود آهن.  سرچ لاءِ لفظ (Rich Keyword Mapping): هر ايموجي سان گڏ پائپ (|) سان ڌار ٿيل لفظ شامل آهن جيئن سرچ ڪرڻ ۾ آساني ٿئي (مثال طور: "مرڪڻ" يا "خوش" لکڻ سان "😀" سرچ ٿيڻ).  ڪيٽيگريز (Categorized): ايموجيز کي انهن جي ڪيٽيگريز جي حساب سان ورهايو ويو آهي (مثال طور: smileys_emotion، people_body).  معياري يونيڪوڊ (Standardized Unicode): هر ايموجي جو يونيڪوڊ (مثال: U+1F600) پڻ ڏنل آهي ته جيئن رينڊرنگ ۾ ڪو مسئلو نه اچي.  📂 ڊيٽا جي بناوٽپروجيڪٽ جي ڊيٽا emojis-data.js فائيل اندر window.EMOJI_DATA نالي ايري (Array) ۾ رکيل آهي.  هر هڪ ايموجي جي ڊيٽا جو ڍانچو ڪجهه هن طرح آهي:  JavaScript{
  "emoji": "😀",                                // ايموجي ڪيريڪٽر
  "code": "U+1F600",                           // يونيڪوڊ هيڪس ڪوڊ
  "category": "smileys_emotion",               // ڪيٽيگري
  "name_en": "grinning face",                  // انگريزي نالو
  "keywords_eng": "cheerful|cheery|face...",   // انگريزي سرچ لفظ (پائپ سان ڌار ٿيل)
  "name_sd": "خوش مزاج",                        // سنڌي نالو
  "keywords_sd": "خوش مزاج|خوش|چهرو|کلڻ|مرڪڻ..." // سنڌي سرچ لفظ (پائپ سان ڌار ٿيل)
}
🚀 استعمال جو طريقوتوهان emojis-data.js فائيل کي سڌو پنهنجي فرنٽ-اينڊ (Front-end) پروجيڪٽ ۾ شامل ڪري سگهو ٿا يا وري ڊيٽابيس ۾ امپورٽ ڪري سگهو ٿا.مثال: سنڌي ۾ ايموجي سرچ ڪرڻJavaScript// سڀ کان پهريان فائيل کي HTML ۾ شامل ڪريو
// <script src="path/to/emojis-data.js"></script>

function searchSindhiEmojis(searchTerm) {
  return window.EMOJI_DATA.filter(item => {
    // پڪ ڪريو ته ايموجي ۾ سنڌي لفظ موجود آهن
    if (!item.keywords_sd) return false; 
    
    const keywords = item.keywords_sd.split('|');
    return keywords.includes(searchTerm) || item.name_sd.includes(searchTerm);
  });
}

// استعمال جو مثال:
const results = searchSindhiEmojis("خوش");
console.log(results); // نتيجي ۾ 😀, 😃 ۽ ٻين ايموجيز جي ڊيٽا ملندي
